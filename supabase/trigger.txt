create or replace function public.handle_new_auth_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  full_name text;
  given_name text;
  family_name text;
  mobile_text text;
begin
  full_name := coalesce(
    new.raw_user_meta_data->>'full_name',
    new.raw_user_meta_data->>'name',
    ''
  );

  given_name := coalesce(
    new.raw_user_meta_data->>'given_name',
    nullif(split_part(full_name, ' ', 1), ''),
    null
  );

  family_name := coalesce(
    new.raw_user_meta_data->>'family_name',
    nullif(trim(replace(full_name, coalesce(given_name, ''), '')), ''),
    null
  );

  mobile_text := coalesce(
    nullif(new.phone, ''),
    nullif(new.raw_user_meta_data->>'phone', ''),
    nullif(new.raw_user_meta_data->>'phone_number', ''),
    nullif(new.raw_user_meta_data->>'mobile', ''),
    null
  );

  insert into public."User" (user_id, email, first_name, last_name, mobile, created_at)
  values (new.id, new.email, given_name, family_name, mobile_text, now())
  on conflict (user_id)
  do update set
    email = excluded.email,
    first_name = coalesce(nullif(public."User".first_name, ''), excluded.first_name),
    last_name  = coalesce(nullif(public."User".last_name, ''),  excluded.last_name),
    mobile     = coalesce(nullif(public."User".mobile, ''),     excluded.mobile);

  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;

create trigger on_auth_user_created
after insert on auth.users
for each row
execute function public.handle_new_auth_user();
